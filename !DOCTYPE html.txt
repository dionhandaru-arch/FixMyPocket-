    <script>
        let currentAuthMode = 'register';
        let currentUser = null;
        let dbUsers = JSON.parse(localStorage.getItem('yf_users') || '{}');

        function formatRupiah(number) {
            return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', maximumFractionDigits: 0 }).format(number);
        }

        window.onload = function() {
            const activeUser = localStorage.getItem('yf_active_user');
            if (activeUser && dbUsers[activeUser]) {
                currentUser = activeUser;
                showDashboard();
            } else {
                showAuth();
            }
        };

        function switchAuthTab(mode) {
            currentAuthMode = mode;
            const regBtn = document.getElementById('tabRegisterBtn');
            const loginBtn = document.getElementById('tabLoginBtn');
            const submitBtn = document.getElementById('authSubmitBtn');

            if (mode === 'register') {
                regBtn.className = "flex-1 py-2 text-sm font-semibold rounded-lg bg-white shadow-sm text-slate-800";
                loginBtn.className = "flex-1 py-2 text-sm font-semibold rounded-lg text-slate-500";
                submitBtn.innerText = "Daftar Akun Baru";
            } else {
                loginBtn.className = "flex-1 py-2 text-sm font-semibold rounded-lg bg-white shadow-sm text-slate-800";
                regBtn.className = "flex-1 py-2 text-sm font-semibold rounded-lg text-slate-500";
                submitBtn.innerText = "Masuk ke Akun";
            }
        }

        function handleAuth(e) {
            e.preventDefault();
            const email = document.getElementById('authEmail').value.trim().toLowerCase();
            const password = document.getElementById('authPassword').value;

            if (currentAuthMode === 'register') {
                if (dbUsers[email]) {
                    alert('Email sudah terdaftar! Silakan login.');
                    return;
                }
                dbUsers[email] = {
                    password: password,
                    status: '',
                    isOnboarded: false,
                    saldoKebutuhan: 0,
                    saldoJajan: 0,
                    saldoTabungan: 0,
                    targetItem: null
                };
                localStorage.setItem('yf_users', JSON.stringify(dbUsers));
                currentUser = email;
                localStorage.setItem('yf_active_user', email);
                showOnboarding();
            } else {
                if (!dbUsers[email] || dbUsers[email].password !== password) {
                    alert('Email atau kata sandi salah!');
                    return;
                }
                currentUser = email;
                localStorage.setItem('yf_active_user', email);
                
                if (!dbUsers[email].isOnboarded) {
                    showOnboarding();
                } else {
                    showDashboard();
                }
            }
        }

        function handleOnboarding(e) {
            e.preventDefault();
            const status = document.getElementById('userStatus').value;
            const income = parseFloat(document.getElementById('initialIncome').value);

            if (!status || isNaN(income)) return;

            let user = dbUsers[currentUser];
            user.status = status;
            user.isOnboarded = true;

            user.saldoKebutuhan = income * 0.5;
            user.saldoJajan = income * 0.3;
            user.saldoTabungan = income * 0.2;

            saveData();
            showDashboard();
        }

        function handleAddIncome(e) {
            e.preventDefault();
            const input = document.getElementById('newIncomeInput');
            const amount = parseFloat(input.value);

            if (isNaN(amount) || amount <= 0) return;

            let user = dbUsers[currentUser];

            user.saldoKebutuhan += (amount * 0.5);
            user.saldoTabungan += (amount * 0.2);
            user.saldoJajan = (amount * 0.3);

            saveData();
            updateDashboardUI();
            input.value = '';
            alert('Pemasukan baru berhasil ditambahkan.');
        }

        function handleSaveTarget(e) {
            e.preventDefault();
            const name = document.getElementById('targetNameInput').value.trim();
            const price = parseFloat(document.getElementById('targetPriceInput').value);

            if (!name || isNaN(price) || price <= 0) return;

            dbUsers[currentUser].targetItem = { name, price };
            saveData();
            updateDashboardUI();
            document.getElementById('targetNameInput').value = '';
            document.getElementById('targetPriceInput').value = '';
        }

        function handleBuyTarget() {
            let user = dbUsers[currentUser];
            if (!user.targetItem) return;

            if (user.saldoTabungan >= user.targetItem.price) {
                user.saldoTabungan -= user.targetItem.price;
                alert(`Pembelian ${user.targetItem.name} berhasil dicatat.`);
                user.targetItem = null;
                saveData();
                updateDashboardUI();
            } else {
                alert('Saldo tabungan belum mencukupi!');
            }
        }

        function handleExpenseNeeds(e) {
            e.preventDefault();
            const desc = document.getElementById('expenseDesc').value.trim();
            const amount = parseFloat(document.getElementById('expenseAmount').value);

            if (!desc || isNaN(amount) || amount <= 0) return;

            let user = dbUsers[currentUser];

            if (user.saldoKebutuhan < amount) {
                alert('Saldo Kebutuhan Pokok tidak mencukupi!');
                return;
            }

            user.saldoKebutuhan -= amount;
            saveData();
            updateDashboardUI();

            document.getElementById('expenseDesc').value = '';
            document.getElementById('expenseAmount').value = '';
            alert(`Pengeluaran "${desc}" sebesar ${formatRupiah(amount)} berhasil dicatat.`);
        }

        function saveData() {
            localStorage.setItem('yf_users', JSON.stringify(dbUsers));
        }

        function showAuth() {
            document.getElementById('authScreen').classList.remove('hidden');
            document.getElementById('onboardingScreen').classList.add('hidden');
            document.getElementById('dashboardScreen').classList.add('hidden');
        }

        function showOnboarding() {
            document.getElementById('authScreen').classList.add('hidden');
            document.getElementById('onboardingScreen').classList.remove('hidden');
            document.getElementById('dashboardScreen').classList.add('hidden');
        }

        function showDashboard() {
            document.getElementById('authScreen').classList.add('hidden');
            document.getElementById('onboardingScreen').classList.add('hidden');
            document.getElementById('dashboardScreen').classList.remove('hidden');
            updateDashboardUI();
        }

        function updateDashboardUI() {
            let user = dbUsers[currentUser];
            if (!user) return;

            document.getElementById('userDisplayEmail').innerText = currentUser;
            document.getElementById('userDisplayStatus').innerText = user.status || 'Pengguna';

            document.getElementById('saldoKebutuhan').innerText = formatRupiah(user.saldoKebutuhan);
            document.getElementById('saldoJajan').innerText = formatRupiah(user.saldoJajan);
            document.getElementById('saldoTabungan').innerText = formatRupiah(user.saldoTabungan);

            const total = user.saldoKebutuhan + user.saldoJajan + user.saldoTabungan;
            document.getElementById('totalBalance').innerText = formatRupiah(total);

            const targetCard = document.getElementById('targetDisplayCard');
            if (user.targetItem) {
                targetCard.classList.remove('hidden');
                document.getElementById('targetItemName').innerText = user.targetItem.name;
                document.getElementById('targetItemPrice').innerText = `Harga: ${formatRupiah(user.targetItem.price)}`;

                const motivationText = document.getElementById('targetMotivationText');
                const badge = document.getElementById('targetStatusBadge');
                const buyBtn = document.getElementById('buyTargetBtn');

                if (user.saldoTabungan >= user.targetItem.price) {
                    motivationText.innerText = "Udah cukup nih!";
                    motivationText.className = "text-xs font-bold text-center py-1 text-emerald-600";
                    badge.innerText = "Cukup";
                    badge.className = "text-[10px] font-bold px-2 py-0.5 rounded-full bg-emerald-100 text-emerald-700";
                    buyBtn.classList.remove('hidden');
                } else {
                    motivationText.innerText = "Yah belum cukup, ayo sedikit lagi!";
                    motivationText.className = "text-xs font-medium text-center py-1 text-amber-600";
                    badge.innerText = "Belum Cukup";
                    badge.className = "text-[10px] font-bold px-2 py-0.5 rounded-full bg-amber-100 text-amber-700";
                    buyBtn.classList.add('hidden');
                }
            } else {
                targetCard.classList.add('hidden');
            }
